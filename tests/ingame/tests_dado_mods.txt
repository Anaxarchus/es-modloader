# The Dado loader's in-game test, run by tests/ingame/run.sh in the loader's
# repository (es-modloader). That script installs this file and the mods in
# tests/ingame/scripts as a plugin ("Dado mod tests") beside the game's own
# integration-test plugin, whose saves and calls it uses, and the loader in the
# test config's modloaders folder. The mods are rebuilt by
# `tools/dado-mods.sh examples`.

test "Dado mods - Hull regen example"
	status active
	description "With the Dado loader installed, the mods load, a test mod halves the flagship's hull and the example mod repairs it."
	sequence
		inject "Three Earthly Barges Save"
		call "Load First Savegame"
		call "Depart"
		apply
			"test: dado probe" = 1
			"dado example: enabled" = 1
			"test: steps to wait" = 200
		label "flying"
		# Empty input to make time pass.
		input
		apply
			"test: steps to wait" = "test: steps to wait" - 1
		branch "flying"
			"test: steps to wait" > 0
		assert
			and
				"mod loaders" >= 1
				"mods loaded" >= 4
				"test: dado damaged" == 1
				"dado example: steps" >= 150
				"dado example: repairs" >= 2
				"test: dado flagship hull" > "test: dado hull start"
